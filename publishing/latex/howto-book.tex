% ---------- Callouts (Key Idea, Watch Out, Try This, Recap, Author Reflection)
% Each callout is set in a box that never splits across pages: if it doesn't
% fit, it moves whole to the next page.
\newcommand\hw@chip[3]{\colorbox{#1}{\strut\sffamily\bfseries\fontsize{7.5}{9}\selectfont\color{#2}#3}}
\newsavebox\hw@calloutbox
\newlength\hw@calloutpad
\newenvironment{hwcallout}[2]{%
  \def\hw@kind{#2}%
  \def\hw@k@watch{watch}\def\hw@k@try{try}\def\hw@k@recap{recap}\def\hw@k@reflection{reflection}%
  \setlength\hw@calloutpad{18pt}%
  \ifx\hw@kind\hw@k@try
    \def\hw@frame##1{{\fboxsep=9pt\fboxrule=0.9pt\fcolorbox{hwSky}{white}{##1}}}%
    \addtolength\hw@calloutpad{1.8pt}%
    \def\hw@chipcmd{\hw@chip{hwSky}{white}{#1}}%
  \else\ifx\hw@kind\hw@k@watch
    \def\hw@frame##1{{\fboxsep=9pt\colorbox{hwMist}{##1}}}%
    \def\hw@chipcmd{\hw@chip{hwGold}{hwNavy}{#1}}%
  \else\ifx\hw@kind\hw@k@recap
    \def\hw@frame##1{{\fboxsep=9pt\colorbox{hwMist}{##1}}}%
    \def\hw@chipcmd{\hw@chip{hwNavy}{white}{#1}}%
  \else\ifx\hw@kind\hw@k@reflection
    \def\hw@frame##1{{\fboxsep=9pt\fboxrule=0.9pt\fcolorbox{hwGold}{white}{##1}}}%
    \addtolength\hw@calloutpad{1.8pt}%
    \def\hw@chipcmd{\hw@chip{hwGold}{hwNavy}{#1}}%
  \else
    \def\hw@frame##1{{\fboxsep=9pt\colorbox{hwMist}{##1}}}%
    \def\hw@chipcmd{\hw@chip{hwAccent}{white}{#1}}%
  \fi\fi\fi\fi
  \par\vspace{0.6\onelineskip}%
  \begin{lrbox}{\hw@calloutbox}%
  \begin{minipage}{\dimexpr\linewidth-\hw@calloutpad\relax}%
  \setlength{\parindent}{0pt}\setlength{\parskip}{0.4\onelineskip}%
  \small\noindent\hw@chipcmd\par\vspace{2pt}}%
 {\par\end{minipage}\end{lrbox}%
  \noindent\hw@frame{\usebox\hw@calloutbox}\par\vspace{0.4\onelineskip}}
